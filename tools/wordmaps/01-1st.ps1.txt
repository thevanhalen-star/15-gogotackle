}

$totalChangedFiles = 0
$totalRealChanges = 0

function Get-LiteralMatchCount {
    param(
        [string]$Text,
        [string]$OldValue
    )

    if ([string]::IsNullOrEmpty($OldValue)) { return 0 }

    $escaped = [regex]::Escape($OldValue)
    return ([regex]::Matches($Text, $escaped)).Count
}

$sortedKeys = $replaceMap.Keys | Sort-Object Length -Descending

Get-ChildItem -Path . -Recurse -File |
Where-Object { $_.Extension.ToLower() -in ".txt", ".html" } |
ForEach-Object {
    $path = $_.FullName

    try {
        $original = [System.IO.File]::ReadAllText($path)
        $working = $original
        $fileChangeCount = 0

        $placeholderMap = [ordered]@{}
        $index = 0

        foreach ($old in $sortedKeys) {
            $new = $replaceMap[$old]

            $count = Get-LiteralMatchCount -Text $working -OldValue $old

            if ($count -gt 0) {
                $placeholder = "__OAI_PLACEHOLDER_$index" + "__"

                while ($working.Contains($placeholder) -or $original.Contains($placeholder)) {
                    $index++
                    $placeholder = "__OAI_PLACEHOLDER_$index" + "__"
                }

                $working = $working.Replace($old, $placeholder)
                $placeholderMap[$placeholder] = $new
                $fileChangeCount += $count
                $index++
            }
        }

        foreach ($ph in $placeholderMap.Keys) {
            $working = $working.Replace($ph, $placeholderMap[$ph])
        }

        if ($working -ne $original) {
            [System.IO.File]::WriteAllText($path, $working)
            $totalChangedFiles++
            $totalRealChanges += $fileChangeCount
            Write-Host "$path : 실제 변경 $fileChangeCount 개"
        }
    }
    catch {
        Write-Host "오류: $path"
    }
}

Write-Host "==============================="
Write-Host "실제 변경된 파일 수 : $totalChangedFiles"
Write-Host "실제 변경 개수 : $totalRealChanges"
